```
import pandas as pd

# Load data from a CSV file
data = pd.read_csv('data.csv')

# Display the first few rows of the DataFrame
print(data.head())

# Display the last few rows of the DataFrame
print(data.tail())

# Display basic information about the DataFrame
print(data.info())

# Display statistical summary of the DataFrame
print(data.describe())
```

### Example 2: Data Selection and Filtering

```python
# Select a specific column
print(data['column_name'])

# Select rows where a condition is met
filtered_data = data[data['column_name'] > 10]

# Select rows and columns using loc
selected_data = data.loc[data['column_name'] > 10, ['column1', 'column2']]
```

### Example 3: Data Manipulation

```python
# Adding a new column
data['new_column'] = data['column1'] + data['column2']

# Dropping a column
data.drop('column_name', axis=1, inplace=True)

# Renaming a column
data.rename(columns={'old_column_name': 'new_column_name'}, inplace=True)
```

### Example 4: Grouping and Aggregation

```python
# Grouping data by a column and aggregating mean
grouped_data = data.groupby('column_name').mean()

# Aggregating data using sum
grouped_data = data.groupby('column_name').sum()
```

### Example 5: Data Visualization

```python
import matplotlib.pyplot as plt

# Plotting a histogram
data['column_name'].hist(bins=10)
plt.title('Histogram of Column Name')
plt.xlabel('Value')
plt.ylabel('Frequency')
plt.show()

# Plotting a line chart
data.plot(x='column1', y='column2', kind='line')
plt.title('Line Chart')
plt.xlabel('Column 1')
plt.ylabel('Column 2')
plt.show()
```
